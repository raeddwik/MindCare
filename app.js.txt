    db.availability
      .find(item => item.date === date);

    ? data.slots

    getAvailableSlots(
      date,

      class="btn secondary"

/* =========================
   TIME FUNCTIONS
========================= */

function toMinutes(time) {

  const parts =
    time.split(":");

  return (
    Number(parts[0]) * 60
    + Number(parts[1])
  );

}


function addMinutes(time, amount) {

  const value =
    toMinutes(time) + amount;

  const hours =
    Math.floor(value / 60);

  const minutes =
    value % 60;

  return (

    String(hours).padStart(2, "0")
    + ":"
    + String(minutes).padStart(2, "0")

  );

}


function overlaps(
  start,
  duration,
  bookedStart,
  bookedDuration
) {

  const a =
    toMinutes(start);

  const b =
    a + duration;

  const c =
    toMinutes(bookedStart);

  const d =
    c + bookedDuration;


  return a < d && c < b;

}


function getAvailableSlots(
  date,
  duration
) {

  const data =
    db.availability
      .find(item => item.date === date);


  if (!data)
    return [];


  return data.slots.filter(time => {

    const conflict =
      db.bookings.some(

        appointment =>

          appointment.date === date
          &&
          overlaps(
            time,
            duration,
            appointment.time,
            appointment.duration
          )

      );


    if (conflict)
      return false;


    if (duration === 60) {

      const secondSlot =
        addMinutes(time, 30);

      return data.slots
        .includes(secondSlot);

    }


    return true;

  });

}

    document

      .value.trim();

    document
      .getElementById("clientEmail")
      .value.trim();

    document
      .getElementById("consent")
      .checked;

      "يرجى إدخال الاسم ورقم الهاتف والموافقة على الشروط."

      new Date().toISOString()

  saveDatabase();


  render(`

/* =========================
   ADD AVAILABILITY
========================= */

function addAvailability() {

  const date =
    document
      .getElementById("availabilityDate")
      .value;


  const start =
    document
      .getElementById("startTime")
      .value;


  const end =
    document
      .getElementById("endTime")
      .value;


  if (!date || !start || !end) {

    alert(
      "يرجى إدخال التاريخ ووقت البداية والنهاية."
    );

    return;

  }


  if (
    toMinutes(end)
    <=
    toMinutes(start)
  ) {

    alert(
      "وقت النهاية يجب أن يكون بعد وقت البداية."
    );

    return;

  }


  const slots = [];


  for (
    let time = toMinutes(start);
    time < toMinutes(end);
    time += 30
  ) {

    slots.push(
      addMinutes("00:00", time)
    );

  }


  let existing =
    db.availability
      .find(item => item.date === date);


  if (existing) {

    existing.slots = [
      ...new Set([
        ...existing.slots,
        ...slots
      ])
    ].sort();

  }

  else {

    db.availability.push({

      date,

      slots

    });

  }


  saveDatabase();

  alert(
    "تمت إضافة المواعيد بنجاح."
  );

  admin();

}

/* =========================
   DATE FORMAT
========================= */

function formatDate(date) {

  return new Date(
    date + "T00:00:00"
  ).toLocaleDateString(
    "ar-EG",
    {
      weekday: "long",
      day: "numeric",
      month: "long"
    }
  );

}

   ROUTER
========================= */

function router() {

  const page =
    location.hash
      .replace("#", "")
      || "home";


  if (page === "book")
    book();

  else if (page === "appointments")
    appointments();

  else if (page === "admin")
    admin();

  else
    home();

}

  "hashchange",

);
